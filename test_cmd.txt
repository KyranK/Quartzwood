# Reset db
Remove-Item quartzwood.db
alembic upgrade head
qw init                  # now seeds default entity/collections/storage
qw list-entities         # should show "Me"
qw list-collections      # should show Main, Trade, Unsorted
qw list-storage          # should show UnsortedBox, TradeBox

# Entity
qw new-entity "Me" --type personal
qw new-entity "LocalShop" --type shop
qw list-entities
qw find-entity --name "Me"
qw update-entity "Me" --new-location "Brisbane"
qw update-entity "LocalShop" --new-name "CardShop"

# Setup
qw new-collection "Main" 
qw new-collection "Trade Stock"
qw new-storage "RedBinder" -c "Main"
qw new-storage "BlueBinder" -c "Main"
qw new-storage "TradeBox" -c "Trade Stock"

# Add cards
qw add 187 MH3 NM -s RedBinder
qw add 13 MH2 NM -s RedBinder -q 4
qw add 189 MH3 LP -s BlueBinder
qw add 13 MH2 NM -s TradeBox -q 2 -f traditional

# List
qw list-entities
qw list-collections
qw list-storage
qw list-cards

# View
qw view-collection Main
qw view-storage RedBinder

# Entity delete tests
qw new-entity "TempEntity"
qw new-collection "TempCollection"
qw rmv-entity "TempEntity"                              # should fail - has collection? no, so should pass
qw rmv-entity "CardShop" --force                        # orphan collections

# Error cases
qw new-entity "Me"                                      # duplicate
qw find-entity --name "Nobody"                          # not found
qw rmv-entity "Nobody"                                  # not found
